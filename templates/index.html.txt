<script>
// Character counter for SMS
function countChars(){
  let msg = document.getElementById('message').value;
  document.getElementById('charCount').innerText = msg.length;
  document.getElementById('smsParts').innerText = Math.ceil(msg.length / 160) || 1;
}

// Read CSV and put numbers in textarea
document.getElementById('csvFile').addEventListener('change', function(e){
  const file = e.target.files[0];
  if(!file) return;
  const reader = new FileReader();
  reader.onload = function(event){
    let numbers = event.target.result.split(/[\n,]/).map(n=>n.trim()).filter(n=>n);
    document.getElementById('to').value = numbers.join(',');
  }
  reader.readAsText(file);
});

// Send SMS function
function sendSMS(){
  let to = document.getElementById('to').value;
  let message = document.getElementById('message').value;
  document.getElementById('result').innerText = "Sending...";

  fetch('/send-sms', {
    method: 'POST',
    headers: {'Content-Type': 'application/json'},
    body: JSON.stringify({to: to, message: message})
  })
 .then(r=>r.json())
 .then(d=>{
    document.getElementById('result').innerText = d.status || d.error;
    if(d.status){ document.getElementById('to').value = ""; document.getElementById('message').value = ""; }
  })
 .catch(err=>{document.getElementById('result').innerText = "Error: " + err});
}
</script>
